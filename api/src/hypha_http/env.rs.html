1//! Boot-time configuration — the substrate half of peregrin's
2//! `env.ts`: bind, auth token, CORS, SSE keep-alive, pairing, node
3//! identity, and the push document. Session-plane tunables and
4//! integration-specific knobs deliberately don't ride `Env` — each
5//! integration parses its own. `Env::parse` fails fast on bad values.
6
7use std::collections::HashMap;
8use std::path::PathBuf;
9use std::time::Duration;
10
11pub const DEFAULT_PORT: u16 = 18787;
12pub const DEFAULT_HOST: &str = "127.0.0.1";
13pub const DEFAULT_SSE_KEEPALIVE_MS: u64 = 15_000;
14/// `POST /api/pair` redeem attempts per client IP: ~40-bit codes make
15/// online guessing infeasible anyway, so these only need to throttle
16/// abuse on nodes exposed off-LAN — not gate legit pairing.
17pub const DEFAULT_PAIR_LIMIT_PER_MIN: u32 = 5;
18pub const DEFAULT_PAIR_LIMIT_PER_DAY: u32 = 20;
19
20const DEFAULT_ORIGINS: [&str; 2] = ["http://localhost:13000", "http://127.0.0.1:13000"];
21
22/// A boot-time configuration error.
23#[derive(Debug, thiserror::Error)]
24#[error("{0}")]
25pub struct EnvError(pub String);
26
27/// Parsed `HYPHA_*` environment — every field the substrate routes read.
28#[derive(Clone, Debug)]
29pub struct Env {
30    /// `HYPHA_PORT` (falls back to the generic `PORT`).
31    pub port: u16,
32    /// `HYPHA_HOST`.
33    pub host: String,
34    /// `HYPHA_TOKEN` — bearer auth; required on non-loopback binds.
35    pub token: Option<String>,
36    /// `HYPHA_HOME` — hypha-owned data dir (node.json, pair-code, tokens, push).
37    pub home: PathBuf,
38    /// `HYPHA_NODE` — stable node identity file.
39    pub node_path: PathBuf,
40    /// `HYPHA_NAME` — display name reported by `GET /api/node`.
41    pub node_name: String,
42    /// `HYPHA_ORIGINS` — CORS allowlist (`*` echoes any origin).
43    pub origins: Vec<String>,
44    /// `HYPHA_SSE_KEEPALIVE_MS` — `/api/events` keep-alive cadence; `0` disables.
45    pub sse_keep_alive: Duration,
46    /// `HYPHA_PAIR_LIMIT_PER_MIN` — `/api/pair` attempts/min per client
47    /// IP (sliding window); `0` disables this window.
48    pub pair_limit_per_min: u32,
49    /// `HYPHA_PAIR_LIMIT_PER_DAY` — `/api/pair` attempts/24h per client
50    /// IP; `0` disables. Both at `0` turns the throttle off entirely.
51    pub pair_limit_per_day: u32,
52    /// `<home>/pair-code` — written by the pairing CLI, consumed on read.
53    pub pair_code_path: PathBuf,
54    /// `<home>/tokens.json` — sha256 hashes of issued pair credentials.
55    pub tokens_path: PathBuf,
56    /// `HYPHA_PUSH`/`HYPHA_PUSH_PATH` — the push config document
57    /// (VAPID keys + subscriptions); `<home>/push.json` by default.
58    pub push_path: PathBuf,
59}

61fn home_dir() -> PathBuf {
62    std::env::var_os("HOME")
63        .or_else(|| std::env::var_os("USERPROFILE"))
64        .map_or_else(|| PathBuf::from("/"), PathBuf::from)
65}
66
67fn hostname_or_default() -> String {
68    nix::unistd::gethostname()
69        .ok()
70        .map(|h| h.to_string_lossy().into_owned())
71        .filter(|h| !h.is_empty())
72        .unwrap_or_else(|| "localhost".to_string())
73}
74
75/// `serve`'s bind guard — loopback binds may go tokenless.
76pub fn is_loopback_host(host: &str) -> bool {
77    host == "localhost" || host == "::1" || host == "[::1]" || host.starts_with("127.")
78}
79
80/// `HYPHA_SSE_KEEPALIVE_MS` style parse: absent → default; a non-finite
81/// or negative value falls back to the default; `0` disables.
82fn ms_or_default(raw: Option<&String>, default: u64) -> u64 {
83    let Some(raw) = raw else { return default };
84    if raw.is_empty() {
85        return default;
86    }
87    match raw.parse::<f64>() {
88        Ok(v) if v.is_finite() && v >= 0.0 => v.floor() as u64,
89        _ => default,
90    }
91}
92
93/// Count-valued knob: absent/unparseable → default (never silently 0 —
94/// a typo must not disable the knob), `0` is the explicit off switch.
95fn count_or_default(raw: Option<&String>, default: u32) -> u32 {
96    raw.filter(|v| !v.is_empty())
97        .and_then(|v| v.parse::<u32>().ok())
98        .unwrap_or(default)
99}

101impl Env {
102    /// Parse from the process environment; fails fast on bad values.
103    ///
104    /// # Errors
105    /// `EnvError` on a bad port or a non-loopback bind without
106    /// `HYPHA_TOKEN`.
107    pub fn parse() -> Result<Self, EnvError> {
108        Self::from_lookup(|key| std::env::var(key).ok())
109    }
110
111    /// Test seam — parse from a key/value map instead of `std::env`.
112    pub fn from_map(vars: &HashMap<String, String>) -> Result<Self, EnvError> {
113        Self::from_lookup(|key| vars.get(key).cloned())
114    }
115
116    /// Shared parser behind [`Env::parse`]/[`Env::from_map`].
117    ///
118    /// # Errors
119    /// See [`Env::parse`].
120    pub fn from_lookup(get: impl Fn(&str) -> Option<String>) -> Result<Self, EnvError> {
121        let home_dir = home_dir();
122
123        // The namespaced var is preferred; the generic `PORT` is the
124        // fallback for bare launches.
125        let raw_port = get("HYPHA_PORT").or_else(|| get("PORT"));
126        let port: u16 = match raw_port.as_deref() {
127            None | Some("") => DEFAULT_PORT,
128            Some(raw) => match raw.parse::<f64>() {
129                Ok(v) if v.fract() == 0.0 && (1.0..=65535.0).contains(&v) => v as u16,
130                _ => {
131                    return Err(EnvError(format!(
132                        "PORT must be an integer between 1 and 65535, got \"{raw}\""
133                    )));
134                }
135            },
136        };
137
138        let origins: Vec<String> = get("HYPHA_ORIGINS")
139            .unwrap_or_default()
140            .split(',')
141            .map(str::trim)
142            .filter(|s| !s.is_empty())
143            .map(str::to_string)
144            .collect();
145        let origins = if origins.is_empty() {
146            DEFAULT_ORIGINS.iter().map(|s| (*s).to_string()).collect()
147        } else {
148            origins
149        };
150
151        let home =
152            get("HYPHA_HOME").map_or_else(|| home_dir.join(".local/share/hypha"), PathBuf::from);
153
154        let host = get("HYPHA_HOST").unwrap_or_else(|| DEFAULT_HOST.to_string());
155        let token = get("HYPHA_TOKEN");
156
157        // The bind guard, enforced at parse so a bad deployment fails
158        // fast at construction rather than mid-request.
159        if !is_loopback_host(&host) && token.as_deref().unwrap_or("").is_empty() {
160            return Err(EnvError(format!(
161                "hypha refuses to bind {host} without HYPHA_TOKEN. Set HYPHA_TOKEN or bind a loopback address (HYPHA_HOST=127.0.0.1)."
162            )));
163        }
164
165        let node_name = get("HYPHA_NAME")
166            .filter(|n| !n.trim().is_empty())
167            .unwrap_or_else(hostname_or_default);
168
169        Ok(Self {
170            port,
171            host,
172            token,
173            home: home.clone(),
174            node_path: get("HYPHA_NODE").map_or_else(|| home.join("node.json"), PathBuf::from),
175            node_name,
176            origins,
177            sse_keep_alive: Duration::from_millis(ms_or_default(
178                get("HYPHA_SSE_KEEPALIVE_MS").as_ref(),
179                DEFAULT_SSE_KEEPALIVE_MS,
180            )),
181            pair_limit_per_min: count_or_default(
182                get("HYPHA_PAIR_LIMIT_PER_MIN").as_ref(),
183                DEFAULT_PAIR_LIMIT_PER_MIN,
184            ),
185            pair_limit_per_day: count_or_default(
186                get("HYPHA_PAIR_LIMIT_PER_DAY").as_ref(),
187                DEFAULT_PAIR_LIMIT_PER_DAY,
188            ),
189            pair_code_path: home.join("pair-code"),
190            tokens_path: home.join("tokens.json"),
191            push_path: get("HYPHA_PUSH")
192                .or_else(|| get("HYPHA_PUSH_PATH"))
193                .map_or_else(|| home.join("push.json"), PathBuf::from),
194        })
195    }
