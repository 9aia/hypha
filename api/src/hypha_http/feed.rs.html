1//! The substrate event feed behind `GET /api/events` — a
2//! process-wide `tokio::sync::broadcast` of kind-tagged payloads that
3//! integrations emit into and SSE connections drain. The shape is
4//! deliberately generic (`kind: &'static str` + opaque `serde_json`
5//! payload): producers name their own event kinds (`session`,
6//! `meta`, `campaign`, …) and consumers branch on the SSE `event:`
7//! field. Integrations bridge their domain events in — e.g. peregrin
8//! maps session-plane events onto this feed.
9//!
10//! `broadcast` already implements the wire semantics: a bounded
11//! per-subscriber queue (`MAX_QUEUED`), oldest-dropped under lag —
12//! a lagged client resyncs through its own channel rather than
13//! blocking the feed.
14
15use serde_json::Value;
16use tokio::sync::broadcast;
17
18/// Bound per subscriber — a client this far behind is resyncing anyway.
19pub const MAX_QUEUED: usize = 500;
20
21/// One feed event: `kind` is the SSE `event:` field (`session`,
22/// `meta`, `heartbeat`, or whatever an integration emits); `payload`
23/// is the `data:` JSON verbatim.
24#[derive(Clone, Debug)]
25pub struct FeedEvent {
26    pub kind: &'static str,
27    pub payload: Value,
28}
29
30/// Process-wide emitter the routes write and SSE connections drain.
31/// `Clone` shares the channel.
32#[derive(Clone)]
33pub struct EventFeed {
34    tx: broadcast::Sender<FeedEvent>,
35}
36
37impl Default for EventFeed {
38    fn default() -> Self {
39        Self::new()
40    }
41}
42
43impl EventFeed {
44    pub fn new() -> Self {
45        let (tx, _) = broadcast::channel(MAX_QUEUED);
46        Self { tx }
47    }
48
49    /// Broadcast `kind`/`payload` to every subscriber. No active
50    /// receivers is not an error.
51    pub fn emit(&self, kind: &'static str, payload: Value) {
52        let _ = self.tx.send(FeedEvent { kind, payload });
53    }
54
55    pub fn subscribe(&self) -> broadcast::Receiver<FeedEvent> {
56        self.tx.subscribe()
57    }
58
59    /// Live SSE subscribers — background work gates on someone
60    /// listening.
61    pub fn subscriber_count(&self) -> usize {
62        self.tx.receiver_count()
63    }
64}

66#[cfg(test)]
67mod tests {
68    #![allow(clippy::unwrap_used)]
69    use serde_json::json;
70
71    use super::*;
72
73    #[tokio::test]
74    async fn emit_reaches_subscribers_in_order() {
75        let feed = EventFeed::new();
76        let mut rx = feed.subscribe();
77        feed.emit("alpha", json!({ "n": 1 }));
78        feed.emit("beta", json!({ "n": 2 }));
79        let first = rx.recv().await.unwrap();
80        assert_eq!(first.kind, "alpha");
81        assert_eq!(first.payload, json!({ "n": 1 }));
82        let second = rx.recv().await.unwrap();
83        assert_eq!(second.kind, "beta");
84    }
85
86    #[tokio::test]
87    async fn emit_without_subscribers_is_not_an_error() {
88        let feed = EventFeed::new();
89        feed.emit("lonely", json!(null));
90        assert_eq!(feed.subscriber_count(), 0);
91    }
92
93    #[tokio::test]
94    async fn lagged_subscriber_survives_the_flood() {
95        let feed = EventFeed::new();
96        let mut rx = feed.subscribe();
97        for i in 0..(MAX_QUEUED * 3) {
98            feed.emit("flood", json!(i));
99        }
100        // However far the subscriber lagged, a marker emitted after
101        // the flood must still arrive — the stream never dies.
102        feed.emit("marker", json!("after"));
103        let mut saw_marker = false;
104        let mut saw_lagged = false;
105        for _ in 0..(MAX_QUEUED * 4) {
106            match rx.recv().await {
107                Ok(event) if event.kind == "marker" => {
108                    saw_marker = true;
109                    break;
110                }
111                Ok(_) => {}
112                Err(broadcast::error::RecvError::Lagged(_)) => saw_lagged = true,
113                Err(broadcast::error::RecvError::Closed) => break,
114            }
115        }
116        assert!(saw_marker, "marker event never arrived");
117        assert!(saw_lagged, "flood never produced a lag signal");
118    }
119}
