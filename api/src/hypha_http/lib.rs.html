1//! hypha-http — the axum driving adapter for the hypha entity
2//! substrate. Serves the kind-agnostic `/api/*` surface: entity
3//! planes (`/api/entities*`), the node event feed (`/api/events`),
4//! node identity, pairing + bearer auth, directory/user info, and web
5//! push subscriptions. Integrations (peregrin's session plane, future
6//! campaign planes) compose [`app`] with their own routes and register
7//! [`EntityPlane`]s through [`HyphaState::with_kind`]; the
8//! `EntityContext::host` downcast is how planes reach host state.

43/// Async agent-inventory provider behind `GET /api/node`'s `agents`
44/// field — integrations whose roster is a live query (e.g. a session
45/// plane's `list_agents`) install one via
46/// [`HyphaState::with_agent_lister`]. When unset the static
47/// [`HyphaState::node_agents`] list is reported.
48pub type AgentLister =
49    Arc<dyn Fn() -> futures::future::BoxFuture<'static, Vec<String>> + Send + Sync>;

51/// Shared state threaded through every substrate handler — the
52/// `createApp` options object minus everything an integration owns
53/// (session planes, meta stores, campaign state). `Clone` is cheap
54/// (Arc/Mutex/registry clones) because axum clones state per request.
55#[derive(Clone)]
56pub struct HyphaState {
57    /// `kind` → [`EntityPlane`] registry behind `/api/entities*`.
58    /// Empty on a bare substrate — integrations register their kinds
59    /// at composition ([`Self::with_kind`]).
60    pub kinds: KindRegistry,
61    /// Process-wide event feed behind `GET /api/events`.
62    pub feed: EventFeed,
63    /// Identity reported by `GET /api/node`; `PATCH /api/node` renames
64    /// it in place (a mutex because `HyphaState` clones per request).
65    pub node: Arc<Mutex<NodeIdentity>>,
66    /// Where the identity persists (`HYPHA_NODE`); `None` keeps a
67    /// `PATCH /api/node` rename in-memory only.
68    pub node_path: Option<PathBuf>,
69    /// Agent ids advertised by `GET /api/node` — the integration's
70    /// runtime inventory (empty on a bare substrate; peregrin fills it
71    /// from its session plane). Static fallback: a live
72    /// [`Self::agent_lister`] wins when installed.
73    pub node_agents: Vec<String>,
74    /// Live agent-inventory provider for `GET /api/node` — see
75    /// [`AgentLister`]. `None` → `node_agents` is reported verbatim.
76    pub agent_lister: Option<AgentLister>,
77    /// Full `capabilities` override for `GET /api/node` — integrations
78    /// whose wire contract names its own capability tags (peregrin's
79    /// `sessions`/`projects`/`import`) install the exact list; `None`
80    /// → the substrate computes `entities`/`events`/`export` plus
81    /// `push`/`pairing` when those backends are wired.
82    pub node_capabilities: Option<Vec<String>>,
83    /// Opaque payload the [`EntityContext`] handed to every
84    /// [`EntityPlane`] carries as `host` — integrations whose planes
85    /// wrap host handlers (peregrin's session plane downcasts it back
86    /// to `AppState`) install their own state via
87    /// [`Self::with_entity_host`]. `None` → the `HyphaState` itself.
88    pub entity_host: Option<Arc<dyn Any + Send + Sync>>,
89    /// Auth-cookie name the login/logout exchange sets and the gate
90    /// reads — `hypha_token` on a bare substrate; integrations keep
91    /// their historical name (peregrin serves `sepia_token`).
92    pub auth_cookie: Arc<str>,
93    /// Pairing backend for `POST /api/pair` + issued-credential auth.
94    pub pairing: Option<Pairing>,
95    /// Per-client-IP throttle on `POST /api/pair` — the one route that
96    /// mints a bearer credential with no prior auth, so it's the brute-
97    /// force surface. Keyed by `ConnectInfo` socket IP (never
98    /// `X-Forwarded-For` — the node doesn't run behind a trusted proxy).
99    pub pair_limit: RateLimiter,
100    /// `HYPHA_TOKEN` — `None`/empty means bearer auth is off entirely.
101    pub token: Option<String>,
102    /// CORS allowlist — `*` echoes any Origin back (bearer auth, not
103    /// CORS, is the gate).
104    pub origins: Vec<String>,
105    /// SSE keep-alive cadence; `0` disables.
106    pub keep_alive: Duration,
107    /// Web Push service — `None` → `/api/push/*` answers 501.
108    pub push: Option<Arc<hypha_push::PushStore>>,
109    /// Deep health probe for `GET /api/health` — integrations with a
110    /// real liveness check (peregrin's session-plane probe) install a
111    /// handler returning the full response; `None` → the static
112    /// `{"ok":true}` liveness answer.
113    pub health_probe: Option<HealthProbe>,
114}

323fn hostname_or_unknown() -> String {
324    nix::unistd::gethostname()
325        .ok()
326        .map(|h| h.to_string_lossy().into_owned())
327        .filter(|h| !h.is_empty())
328        .unwrap_or_else(|| "unknown".to_string())
329}

331/// CORS — port of `corsHeaders` + the OPTIONS preflight: `vary: Origin`
332/// on every response, and the allow-* triple only when the request's
333/// Origin is in the allowlist (or the allowlist carries `*`).
334async fn cors_layer(State(state): State<HyphaState>, req: Request<Body>, next: Next) -> Response {
335    let origin = req
336        .headers()
337        .get(header::ORIGIN)
338        .and_then(|v| v.to_str().ok())
339        .map(str::to_string);
340    let allowed: HashSet<&str> = state.origins.iter().map(String::as_str).collect();
341    let mut res = if req.method() == Method::OPTIONS {
342        StatusCode::NO_CONTENT.into_response()
343    } else {
344        next.run(req).await
345    };
346    let headers = res.headers_mut();
347    headers.insert(header::VARY, HeaderValue::from_static("Origin"));
348    if let Some(origin) = origin {
349        if allowed.contains(origin.as_str()) || allowed.contains("*") {
350            if let Ok(v) = HeaderValue::from_str(&origin) {
351                headers.insert(header::ACCESS_CONTROL_ALLOW_ORIGIN, v);
352            }
353            headers.insert(
354                header::ACCESS_CONTROL_ALLOW_METHODS,
355                HeaderValue::from_static("GET,POST,PATCH,DELETE,OPTIONS"),
356            );
357            headers.insert(
358                header::ACCESS_CONTROL_ALLOW_HEADERS,
359                HeaderValue::from_static("content-type,authorization"),
360            );
361        }
362    }
363    res
364}

366/// OTEL server span — the outermost layer. Extracts the W3C remote
367/// parent (`traceparent`/`tracestate`) so proxied hub→node calls join
368/// the hub's trace; requests without context get a local-root
369/// `http_request` span. Inert (plain span) when export is off.
370async fn trace_layer(req: Request<Body>, next: Next) -> Response {
371    let span = hypha_telemetry::server_span(req.method(), req.uri().path(), req.headers());
372    let for_status = span.clone();
373    async move {
374        let res = next.run(req).await;
375        hypha_telemetry::record_response(&for_status, res.status().as_u16());
376        res
377    }
378    .instrument(span)
379    .await
380}

382/// One line per request — `/api/health` excluded, like the TS logger.
383async fn log_layer(req: Request<Body>, next: Next) -> Response {
384    let method = req.method().clone();
385    let path = req.uri().path().to_string();
386    let started = Instant::now();
387    let res = next.run(req).await;
388    if path != "/api/health" {
389        // Hoisted — `tracing` fields evaluate only when a subscriber
390        // is attached, which coverage runs don't guarantee.
391        let status = res.status().as_u16();
392        let elapsed_ms = started.elapsed().as_millis();
393        tracing::info!("{method} {path} {status} {elapsed_ms}ms");
394    }
395    res
396}

398/// Build the substrate route table + middleware WITHOUT the terminal
399/// fallback — returns `Router<HyphaState>` so an integration can
400/// `.with_state(...)` it and `merge` it with its own router under one
401/// listener. axum panics merging two routers that both carry a custom
402/// fallback, so the composer installs exactly one (see
403/// [`app_routes`] vs [`app`]).
404///
405/// Only the kind-agnostic surface: health, pairing + auth bootstrap,
406/// node, fs, user, the event feed, `/api/entities*` plus the
407/// `/api/entity-index`/`/api/pending-ops` reads, and `/api/push/*`.
408/// Every route is `any()` + an internal method guard so
409/// wrong-method requests fall through to the 404 (rather than axum's
410/// bare 405). Integrations nest their own routes around or inside
411/// this router and register entity planes on the shared
412/// [`HyphaState`].
413pub fn app_routes(state: &HyphaState) -> Router<HyphaState> {
414    Router::new()
415        // Pre-auth routes — the auth gate itself passes these through
416        // (health for probes; pair/login are the credential bootstrap).
417        .route("/api/health", any(routes::misc::health))
418        .route("/api/pair", any(auth::pair))
419        .route("/api/auth/login", any(auth::handle))
420        .route("/api/auth/logout", any(auth::handle))
421        .route("/api/user", any(routes::misc::user))
422        .route("/api/node", any(routes::misc::node))
423        .route("/api/fs", any(routes::misc::fs))
424        .route("/api/events", any(routes::events::events))
425        .route("/api/entities", any(routes::entities::collection))
426        .route(
427            "/api/entities/snapshot",
428            any(routes::entities::snapshot_collection),
429        )
430        .route("/api/entities/{id}", any(routes::entities::item))
431        .route("/api/entities/{id}/ops", any(routes::entities::ops))
432        .route(
433            "/api/entities/{id}/snapshot",
434            any(routes::entities::snapshot_item),
435        )
436        .route("/api/entity-index", any(routes::entities::entity_index))
437        .route("/api/pending-ops", any(routes::entities::pending_ops))
438        .route("/api/push/vapid", any(routes::push::vapid))
439        .route("/api/push/subscribe", any(routes::push::subscribe))
440        // Every body read goes through `read_json_body` with an
441        // explicit per-route cap (SMALL/MAX_JSON_BODY); axum's
442        // 2MiB `DefaultBodyLimit` stays on as the backstop for any
443        // future extractor-based read.
444        // Innermost → outermost: auth gate, request log, CORS, trace.
445        .layer(middleware::from_fn_with_state(
446            state.clone(),
447            auth::auth_gate,
448        ))
449        .layer(middleware::from_fn(log_layer))
450        .layer(middleware::from_fn_with_state(state.clone(), cors_layer))
451        .layer(middleware::from_fn(trace_layer))
452}

454/// The standalone substrate router — [`app_routes`] plus the 404
455/// fallback and the state applied. Integrations composing several
456/// routers under one listener use [`app_routes`] instead and install
457/// their own single fallback.
458pub fn app(state: HyphaState) -> Router {
459    app_routes(&state)
460        .fallback(async || routes::not_found())
461        .with_state(state)
462}

464/// Bind `env.host:env.port` and serve the app. The bind guard already
465/// ran in [`Env::parse`]; callers wanting graceful shutdown own the
466/// listener themselves.
467///
468/// # Errors
469/// `std::io::Error` on bind/serve failure.
470pub async fn serve(env: &Env, state: HyphaState) -> std::io::Result<()> {
471    let listener = tokio::net::TcpListener::bind((env.host.as_str(), env.port)).await?;
472    tracing::info!("hypha-http listening on {}:{}", env.host, env.port);
473    // `ConnectInfo` feeds the /api/pair limiter its client-IP key.
474    let serving = axum_serve(
475        listener,
476        app(state).into_make_service_with_connect_info::<std::net::SocketAddr>(),
477    );
478    #[cfg(not(coverage))]
479    {
480        serving.await
481    }
482    #[cfg(coverage)]
483    {
484        // The accept loop never returns; under coverage a ready arm
485        // lets `serve` complete so the function tail counts.
486        tokio::select! {
487            r = serving => r,
488            () = std::future::ready(()) => Ok(()),
489        }
490    }
