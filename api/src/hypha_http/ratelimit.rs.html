1//! In-memory per-IP sliding-window rate limiting — guards the
2//! credential-bootstrapping `POST /api/pair` (the one route that mints
3//! a bearer token without prior auth) against online code guessing.
4//! No deps: a bounded `HashMap<IpAddr, …>` of attempt timestamps.
5
6use std::collections::{HashMap, VecDeque};
7use std::net::IpAddr;
8use std::sync::{Arc, Mutex, PoisonError};
9use std::time::{Duration, Instant};
10
11/// One window rule: at most `max` hits inside `window`. `max == 0`
12/// (or an empty quota set) disables enforcement.
13#[derive(Clone, Copy, Debug)]
14pub struct Quota {
15    pub max: u32,
16    pub window: Duration,
17}
18
19/// Default cap on tracked IPs — memory stays bounded even when an
20/// attacker sprays source addresses; at capacity the idlest entry is
21/// evicted (expired-hit entries first, then least-recently-seen).
22pub const DEFAULT_MAX_KEYS: usize = 4096;
23
24struct Entry {
25    /// Recorded attempt times, oldest → newest; pruned past the
26    /// longest quota window.
27    hits: VecDeque<Instant>,
28    /// Last touch — the eviction tiebreak when the table is full.
29    last_seen: Instant,
30}
31
32/// A multi-window, per-key limiter. `check` records admitted attempts;
33/// rejected ones don't extend the window, so `Retry-After` is honest —
34/// a flooding IP drains its quota rather than starving forever.
35///
36/// `Clone` shares the table (`AppState` clones per request).
37#[derive(Clone)]
38pub struct RateLimiter {
39    quotas: Vec<Quota>,
40    max_keys: usize,
41    entries: Arc<Mutex<HashMap<IpAddr, Entry>>>,
42}
43
44impl RateLimiter {
45    /// Enforce `quotas` (sorted internally; empty/all-zero = unlimited).
46    /// `max_keys` bounds the per-IP table; clamped to ≥1.
47    pub fn new(quotas: impl IntoIterator<Item = Quota>, max_keys: usize) -> Self {
48        let mut quotas: Vec<Quota> = quotas
49            .into_iter()
50            .filter(|q| q.max > 0 && q.window > Duration::ZERO)
51            .collect();
52        quotas.sort_by_key(|q| q.window);
53        Self {
54            quotas,
55            max_keys: max_keys.max(1),
56            entries: Arc::new(Mutex::new(HashMap::new())),
57        }
58    }
59
60    /// The `POST /api/pair` quotas: `per_min` attempts/minute plus
61    /// `per_day` attempts/24h per client IP (`0` disables each window).
62    pub fn pair(per_min: u32, per_day: u32) -> Self {
63        Self::new(
64            [
65                Quota {
66                    max: per_min,
67                    window: Duration::from_secs(60),
68                },
69                Quota {
70                    max: per_day,
71                    window: Duration::from_secs(86_400),
72                },
73            ],
74            DEFAULT_MAX_KEYS,
75        )
76    }
77
78    /// The longest quota window — hits older than this are dead weight.
79    fn horizon(&self) -> Duration {
80        self.quotas.last().map_or(Duration::ZERO, |q| q.window)
81    }
82
83    /// Record an attempt for `key`: `Ok` admits it, `Err(retry_after)`
84    /// is the wait until enough in-window hits age out.
85    pub fn check(&self, key: IpAddr) -> Result<(), Duration> {
86        self.check_at(key, Instant::now())
87    }
88
89    /// The clock-injectable core of [`Self::check`].
90    pub fn check_at(&self, key: IpAddr, now: Instant) -> Result<(), Duration> {
91        if self.quotas.is_empty() {
92            return Ok(());
93        }
94        let horizon = self.horizon();
95        let mut entries = self.entries.lock().unwrap_or_else(PoisonError::into_inner);
96        if let Some(entry) = entries.get_mut(&key) {
97            entry.last_seen = now;
98            prune(&mut entry.hits, now, horizon);
99            if let Some(wait) = retry_after(&entry.hits, &self.quotas, now) {
100                return Err(wait);
101            }
102            entry.hits.push_back(now);
103            return Ok(());
104        }
105        if entries.len() >= self.max_keys {
106            evict(&mut entries, now, horizon, self.max_keys);
107        }
108        entries.insert(
109            key,
110            Entry {
111                hits: VecDeque::from([now]),
112                last_seen: now,
113            },
114        );
115        Ok(())
116    }
117}
118
119/// Drop hits older than `horizon` — they can't count against any quota.
120fn prune(hits: &mut VecDeque<Instant>, now: Instant, horizon: Duration) {
121    while hits
122        .front()
123        .is_some_and(|t| now.duration_since(*t) >= horizon)
124    {
125        hits.pop_front();
126    }
127}
128
129/// `None` under every quota; `Some(wait)` when one or more windows are
130/// saturated — the longest wait among the violated windows.
131fn retry_after(hits: &VecDeque<Instant>, quotas: &[Quota], now: Instant) -> Option<Duration> {
132    let mut wait = Duration::ZERO;
133    let mut limited = false;
134    for quota in quotas {
135        // Hits are chronological: find the first still inside this window.
136        let first_in = hits
137            .iter()
138            .position(|t| now.duration_since(*t) < quota.window)
139            .unwrap_or(hits.len());
140        let count = hits.len() - first_in;
141        if count >= quota.max as usize {
142            // Admit again once enough of the oldest in-window hits age out.
143            let pivot = hits[first_in + (count - quota.max as usize)];
144            let w = pivot
145                .checked_add(quota.window)
146                .map_or(quota.window, |until| until.saturating_duration_since(now));
147            wait = wait.max(w);
148            limited = true;
149        }
150    }
151    limited.then_some(wait)
152}
153
154/// Table at capacity: drop entries with no live hits first; if none are
155/// idle, evict the least-recently-seen (LRU-ish). Runs only when an
156/// attacker is already spraying keys, so the linear scans are fine.
157fn evict(entries: &mut HashMap<IpAddr, Entry>, now: Instant, horizon: Duration, max_keys: usize) {
158    entries.retain(|_, e| {
159        prune(&mut e.hits, now, horizon);
160        !e.hits.is_empty()
161    });
162    if entries.len() >= max_keys {
163        if let Some(oldest) = entries
164            .iter()
165            .min_by_key(|(_, e)| e.last_seen)
166            .map(|(k, _)| *k)
167        {
168            entries.remove(&oldest);
169        }
170    }
171}

173#[cfg(test)]
174#[allow(clippy::unwrap_used)]
175mod tests {
176    use std::net::Ipv4Addr;
177
178    use super::*;
179
180    const MIN: Duration = Duration::from_secs(60);
181    const DAY: Duration = Duration::from_secs(86_400);
182
183    fn ip(last: u8) -> IpAddr {
184        IpAddr::V4(Ipv4Addr::new(10, 0, 0, last))
185    }
186
187    #[test]
188    fn admits_up_to_max_then_limits() {
189        let limiter = RateLimiter::new(
190            [Quota {
191                max: 2,
192                window: MIN,
193            }],
194            16,
195        );
196        let now = Instant::now();
197        assert!(limiter.check_at(ip(1), now).is_ok());
198        assert!(limiter.check_at(ip(1), now).is_ok());
199        let err = limiter.check_at(ip(1), now).unwrap_err();
200        assert_eq!(err, MIN, "wait until the oldest hit ages out");
201    }
202
203    #[test]
204    fn window_slides() {
205        let limiter = RateLimiter::new(
206            [Quota {
207                max: 2,
208                window: MIN,
209            }],
210            16,
211        );
212        let t0 = Instant::now();
213        assert!(limiter.check_at(ip(1), t0).is_ok());
214        assert!(
215            limiter
216                .check_at(ip(1), t0 + Duration::from_secs(30))
217                .is_ok()
218        );
219        // Inside the window: limited, retry after the oldest hit exits.
220        let err = limiter
221            .check_at(ip(1), t0 + Duration::from_secs(45))
222            .unwrap_err();
223        assert_eq!(err, Duration::from_secs(15));
224        // The oldest hit aged out: the window slid.
225        assert!(limiter.check_at(ip(1), t0 + MIN).is_ok());
226    }

228    #[test]
229    fn quotas_are_independent() {
230        // 5/min AND 20/day — the tighter window limits first.
231        let limiter = RateLimiter::new(
232            [
233                Quota {
234                    max: 5,
235                    window: MIN,
236                },
237                Quota {
238                    max: 20,
239                    window: DAY,
240                },
241            ],
242            16,
243        );
244        let mut now = Instant::now();
245        // Four windows of 5 → 20 hits; the 5th minute is capped by the day.
246        for _ in 0..4 {
247            for _ in 0..5 {
248                assert!(limiter.check_at(ip(1), now).is_ok());
249            }
250            now += MIN;
251        }
252        let err = limiter.check_at(ip(1), now).unwrap_err();
253        assert!(err > Duration::ZERO && err <= DAY);
254    }

256    #[test]
257    fn keys_are_independent() {
258        let limiter = RateLimiter::new(

271    #[test]
272    fn rejected_attempts_do_not_extend_the_window() {
273        let limiter = RateLimiter::new(
274            [Quota {
275                max: 1,
276                window: MIN,
277            }],
278            16,
279        );
280        let t0 = Instant::now();
281        assert!(limiter.check_at(ip(1), t0).is_ok());
282        // Hammering inside the window keeps 429ing without recording —
283        // the wait never grows past the original hit's expiry.
284        for s in [10u64, 20, 30] {
285            let err = limiter
286                .check_at(ip(1), t0 + Duration::from_secs(s))
287                .unwrap_err();
288            assert_eq!(err, MIN.checked_sub(Duration::from_secs(s)).unwrap());
289        }
290        assert!(limiter.check_at(ip(1), t0 + MIN).is_ok());
291    }

293    #[test]
294    fn table_is_bounded_and_evicts_idle() {
295        let limiter = RateLimiter::new(
296            [Quota {
297                max: 5,
298                window: MIN,
299            }],
300            3,
301        );
302        let t0 = Instant::now();
303        // Distinct last_seen so the LRU pick is deterministic.
304        for i in 1..=3u8 {
305            assert!(
306                limiter
307                    .check_at(ip(i), t0 + Duration::from_secs(u64::from(i)))
308                    .is_ok()
309            );
310        }
311        // A 4th key at capacity: all entries' hits are live, so the
312        // least-recently-seen (key 1) is evicted — its limit resets.
313        assert!(limiter.check_at(ip(4), t0 + Duration::from_secs(4)).is_ok());
314        let entries = limiter.entries.lock().unwrap();
315        assert_eq!(entries.len(), 3);
316        assert!(entries.get(&ip(1)).is_none(), "oldest key evicted");
317    }

319    #[test]
320    fn expired_entries_are_evicted_before_live_ones() {
321        let limiter = RateLimiter::new(
322            [Quota {
323                max: 5,
324                window: MIN,
325            }],
326            2,
327        );
328        let t0 = Instant::now();
329        assert!(limiter.check_at(ip(1), t0).is_ok());
330        assert!(limiter.check_at(ip(2), t0 + MIN).is_ok());
331        // ip(1)'s hits are stale; at capacity it evicts, not live ip(2).
332        assert!(limiter.check_at(ip(3), t0 + MIN).is_ok());
333        let entries = limiter.entries.lock().unwrap();
334        assert!(entries.get(&ip(1)).is_none());
335        assert!(entries.get(&ip(2)).is_some());
336    }
