1//! `GET /api/events` — the substrate event feed (the
2//! `routes/events.ts` shape): each connection drains a bounded
3//! subscription; a client that stops draining is terminated rather
4//! than wedging the feed. Heartbeats ride the stream as real
5//! `heartbeat` events, not comment pings. Event kinds are whatever
6//! the registered integrations emit — the route is kind-agnostic.
7
8use std::convert::Infallible;
9
10use axum::body::Body;
11use axum::extract::{Request, State};
12use axum::http::Method;
13use axum::response::sse::{Event, Sse};
14use axum::response::{IntoResponse, Response};
15use futures::stream;
16use serde_json::json;
17use tokio::sync::broadcast;
18
19use crate::HyphaState;
20use crate::routes::{not_found, now_ms_f64};
21
22/// `event: <kind>` + `data: <json>` — the wire `frame()` shape.
23fn frame(kind: &'static str, payload: &serde_json::Value) -> Event {
24    Event::default()
25        .event(kind)
26        .data(serde_json::to_string(payload).unwrap_or_else(|_| "null".into()))
27}
28
29/// `GET /api/events` — SSE over [`crate::EventFeed`]. EventSource
30/// clients authenticate via `?access_token` (honored by the auth gate
31/// on this path only).
32pub async fn events(
33    State(state): State<HyphaState>,
34    method: Method,
35    _req: Request<Body>,
36) -> Response {
37    if method != Method::GET {
38        return not_found();
39    }
40    let mut sub = state.feed.subscribe();
41    let (tx, rx) = tokio::sync::mpsc::channel::<Event>(256);
42    let keep_alive = state.keep_alive;
43    tokio::spawn(async move {
44        let mut heartbeat = (keep_alive > std::time::Duration::ZERO).then(|| {
45            // `setInterval` semantics — the first beat lands after one
46            // interval, not immediately.
47            tokio::time::interval_at(tokio::time::Instant::now() + keep_alive, keep_alive)
48        });
49        loop {
50            tokio::select! {
51                event = sub.recv() => match event {
52                    Ok(event) => {
53                        if tx.try_send(frame(event.kind, &event.payload)).is_err() {
54                            return;
55                        }
56                    }
57                    // A lagged subscriber resyncs on its side; keep the
58                    // stream alive (the feed's queue bound already
59                    // dropped the oldest).
60                    Err(broadcast::error::RecvError::Lagged(_)) => {},
61                    Err(broadcast::error::RecvError::Closed) => return,
62                },
63                _ = async {
64                    match heartbeat.as_mut() {
65                        Some(h) => h.tick().await,
66                        None => std::future::pending().await,
67                    }
68                } => {
69                    if tx
70                        .try_send(frame("heartbeat", &json!({ "ts": now_ms_f64() })))
71                        .is_err()
72                    {
73                        return;
74                    }
75                }
76            }
77        }
78    });
79    let stream = stream::unfold(rx, |mut rx| async move {
80        rx.recv().await.map(|e| (Ok::<_, Infallible>(e), rx))
81    });
82    let mut res = Sse::new(stream).into_response();
83    res.headers_mut().insert(
84        axum::http::header::CONNECTION,
85        axum::http::HeaderValue::from_static("keep-alive"),
86    );
87    res
88}
