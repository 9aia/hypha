22/// `{error: message}` JSON body at `status`.
23pub fn json_response(body: Value, status: StatusCode) -> Response {
24    (status, axum::Json(body)).into_response()
25}
26
27pub fn ok() -> Response {
28    json_response(serde_json::json!({ "ok": true }), StatusCode::OK)
29}

31/// Terminal handler — unknown paths fall through to

42/// Small bodies — pair codes, login tokens, renames, op submissions,
43/// push subscriptions. The biggest legit field anywhere on these
44/// routes is small; 64KiB is generous.
45pub const SMALL_JSON_BODY: usize = 64 * 1024;
46
47/// Hard ceiling — `POST /api/entities/snapshot` carries a whole
48/// exported snapshot; nothing else legitimately approaches this.
49pub const MAX_JSON_BODY: usize = 32 * 1024 * 1024;
50
51/// Read a JSON body with an explicit byte cap: empty body → `None`;
52/// parse failure → the boxed 400 `Invalid JSON body` response; over
53/// `limit` bytes → 413 (`Box` keeps the `Err` variant small for
54/// `clippy::result_large_err`).
55pub async fn read_json_body(
56    req: Request<Body>,
57    limit: usize,
58) -> Result<Option<Value>, Box<Response>> {
59    let bytes = axum::body::to_bytes(req.into_body(), limit)
60        .await
61        .map_err(|e| {
62            if std::error::Error::source(&e)
63                .is_some_and(<dyn std::error::Error>::is::<http_body_util::LengthLimitError>)
64            {
65                Box::new(payload_too_large())
66            } else {
67                Box::new(invalid_json())
68            }
69        })?;
70    read_json_bytes(&bytes)
71}
72
73fn payload_too_large() -> Response {
74    json_response(
75        serde_json::json!({ "error": "Request body too large" }),
76        StatusCode::PAYLOAD_TOO_LARGE,
77    )
78}
79
80fn invalid_json() -> Response {
81    json_response(
82        serde_json::json!({ "error": "Invalid JSON body" }),
83        StatusCode::BAD_REQUEST,
84    )
85}

87pub fn read_json_bytes(bytes: &Bytes) -> Result<Option<Value>, Box<Response>> {
88    let text = String::from_utf8_lossy(bytes);
89    if text.trim().is_empty() {
90        return Ok(None);
91    }
92    serde_json::from_str::<Value>(text.trim())
93        .map(Some)
94        .map_err(|_| Box::new(invalid_json()))
95}

97/// `url.searchParams` — a decoded name→value map.
98pub fn query_params(query: Option<&str>) -> HashMap<String, String> {
99    let Some(query) = query else {
100        return HashMap::new();
101    };
102    form_urlencoded::parse(query.as_bytes())
103        .map(|(k, v)| (k.into_owned(), v.into_owned()))
104        .collect()
105}
106
107pub fn query_param(query: Option<&str>, key: &str) -> Option<String> {
108    query.and_then(|q| {
109        form_urlencoded::parse(q.as_bytes())
110            .find(|(k, _)| k == key)
111            .map(|(_, v)| v.into_owned())
112    })
113}

115/// `Date.now()` — epoch milliseconds.
116pub fn now_ms_f64() -> f64 {
117    std::time::SystemTime::now()
118        .duration_since(std::time::UNIX_EPOCH)
119        .map_or(0.0, |d| d.as_secs_f64() * 1000.0)
120}

127    #[test]
128    fn respond_and_helper_arms() {
129        // Query helpers.
130        assert!(query_params(None).is_empty());
131        assert_eq!(query_params(Some("a=1&b=2")).get("b").unwrap(), "2");
132        assert_eq!(query_param(Some("a=1"), "a").as_deref(), Some("1"));
133        assert_eq!(query_param(None, "a"), None);
134        // Error-response shapes.
135        let _ = invalid_json();
136        let _ = payload_too_large();
137        let _ = ok();
138        let _ = not_found();
139    }
140
141    #[tokio::test]
142    async fn read_json_body_stream_error_is_invalid() {
143        // A body that errors mid-stream (not a length limit) → the
144        // `invalid_json` arm of the `to_bytes` map_err.
145        let body = Body::from_stream(futures::stream::once(async {
146            Err::<axum::body::Bytes, std::io::Error>(std::io::Error::other("boom"))
147        }));
148        let req = Request::builder()
149            .method("POST")
150            .uri("/x")
151            .body(body)
152            .unwrap();
153        let res = read_json_body(req, SMALL_JSON_BODY).await;
154        assert!(res.is_err());
155    }
156
157    #[test]
158    fn read_json_bytes_arms() {
159        assert!(read_json_bytes(&Bytes::new()).unwrap().is_none());
160        assert!(
161            read_json_bytes(&Bytes::from_static(b"  "))
162                .unwrap()
163                .is_none()
164        );
165        assert!(read_json_bytes(&Bytes::from_static(b"{")).is_err());
166        let v = read_json_bytes(&Bytes::from_static(br#"{"a":1}"#)).unwrap();
167        assert_eq!(v.unwrap()["a"], 1);
168    }
