1//! Where the hub keeps node bearer tokens between restarts.
2//!
3//! `HYPHA_NODES` `@token` / `HYPHA_NODE_TOKEN` are explicit config and
4//! always win; an env token also seeds an empty store slot at boot —
5//! the migration path off plaintext env (drop `@token` after one boot,
6//! the token keeps working from the store). A stored credential is
7//! never overwritten by env config; rotation goes through re-pairing.
8//! Nodes listed without a token — and credentials minted by
9//! the pair flow — resolve and persist through a [`NodeCredentials`]
10//! store: the OS keyring (Secret Service on Linux/*BSD, Keychain on
11//! macOS, Credential Manager on Windows) when reachable, else
12//! `credentials.json` (0600) under the hub home — the documented
13//! headless path. `HYPHA_CREDENTIAL_STORE=auto|keyring|file` picks the
14//! backend; `auto` (default) probes the keyring once at boot.
15//!
16//! Keyring ops are blocking (a dbus round-trip on Linux) — async
17//! callers go through `tokio::task::spawn_blocking`, same as `ureq`.
18
19use std::collections::BTreeMap;
20use std::path::{Path, PathBuf};
21use std::sync::{Arc, Mutex};
22
23/// The keyring service; the account is `node/{id}`.
24const KEYRING_SERVICE: &str = "hypha-hub";
25/// `HYPHA_CREDENTIAL_STORE` — `auto` (default), `keyring`, `file`.
26pub const STORE_ENV: &str = "HYPHA_CREDENTIAL_STORE";
27/// File-store name under the hub home (`$HYPHA_HOME/hub/`).
28pub const FILE_NAME: &str = "credentials.json";
29
30/// Node-token persistence. Errors are strings — the surfaces that call
31/// this (hub boot, `NodeApi::pair`) report plain messages.
32pub trait NodeCredentials: Send + Sync {
33    /// The stored token for `node_id`; `Ok(None)` = never stored.
34    fn get(&self, node_id: &str) -> Result<Option<String>, String>;
35    /// Persist `token` under `node_id` (overwrite).
36    fn set(&self, node_id: &str, token: &str) -> Result<(), String>;
37    /// Drop a stored credential (unpair/rotate). Idempotent — deleting
38    /// a missing entry is `Ok`.
39    fn delete(&self, node_id: &str) -> Result<(), String>;
40    /// Which backend is live — for logs and docs, not logic.
41    fn backend(&self) -> &'static str;
42}

44/* ---- OS keyring ------------------------------------------------------*/
45
46/// `Entry::new` builds the crate's *mock* store on platforms where no
47/// real backend feature compiled — it "works" but persists nothing, so
48/// the probe only trusts platforms we enabled a backend for.
49///
50/// `Ok(())` = the OS keyring answered (a miss counts — `NoEntry` is a
51/// healthy response); `Err` = no reachable backend.
52#[cfg(any(
53    target_os = "linux",
54    target_os = "freebsd",
55    target_os = "openbsd",
56    target_os = "macos",
57    target_os = "ios",
58    target_os = "windows"
59))]
60pub fn keyring_probe() -> Result<(), String> {
61    let entry = keyring::Entry::new(KEYRING_SERVICE, "hypha-hub-probe")
62        .map_err(|e| format!("keyring: {e}"))?;
63    match entry.get_password() {
64        // Reachable backend — `NoEntry` is a healthy answer.
65        Ok(_) | Err(keyring::Error::NoEntry) => Ok(()),
66        Err(e) => Err(format!("keyring: {e}")),
67    }
68}
69
70/// `Err` — no real backend was compiled in for this platform (see the
71/// supported-platforms arm above).
72#[cfg(not(any(
73    target_os = "linux",
74    target_os = "freebsd",
75    target_os = "openbsd",
76    target_os = "macos",
77    target_os = "ios",
78    target_os = "windows"
79)))]
80pub fn keyring_probe() -> Result<(), String> {
81    Err("no OS keyring backend on this platform".to_string())
82}

84/// OS keyring — Secret Service / Keychain / Credential Manager.
85pub struct KeyringCredentials;
86
87impl KeyringCredentials {
88    fn entry(node_id: &str) -> Result<keyring::Entry, String> {
89        keyring::Entry::new(KEYRING_SERVICE, &format!("node/{node_id}"))
90            .map_err(|e| format!("keyring: {e}"))
91    }
92}
93
94impl NodeCredentials for KeyringCredentials {
95    fn get(&self, node_id: &str) -> Result<Option<String>, String> {
96        match Self::entry(node_id)?.get_password() {
97            Ok(token) => Ok(Some(token)),
98            Err(keyring::Error::NoEntry) => Ok(None),
99            Err(e) => Err(format!("keyring: {e}")),
100        }
101    }
102
103    fn set(&self, node_id: &str, token: &str) -> Result<(), String> {
104        Self::entry(node_id)?
105            .set_password(token)
106            .map_err(|e| format!("keyring: {e}"))
107    }
108
109    fn delete(&self, node_id: &str) -> Result<(), String> {
110        match Self::entry(node_id)?.delete_credential() {
111            Ok(()) | Err(keyring::Error::NoEntry) => Ok(()),
112            Err(e) => Err(format!("keyring: {e}")),
113        }
114    }
115
116    fn backend(&self) -> &'static str {
117        "os-keyring"
118    }
119}

121/* ---- file fallback ----------------------------------------------------*/
122
123/// `credentials.json` — `{node_id: token}` written 0600 via tmp+rename.
124/// The headless path: no Secret Service means no keyring, and a 0600
125/// file under `$HYPHA_HOME` is exactly as protected as the env files
126/// carrying `HYPHA_NODES` today.
127pub struct FileCredentials {
128    path: PathBuf,
129    /// Serializes read-modify-write in `set`.
130    write: Mutex<()>,
131}
132
133impl FileCredentials {
134    pub fn new(path: PathBuf) -> Self {
135        Self {
136            path,
137            write: Mutex::new(()),
138        }
139    }
140
141    pub fn path(&self) -> &Path {
142        &self.path
143    }
144
145    fn load(&self) -> Result<BTreeMap<String, String>, String> {
146        match std::fs::read_to_string(&self.path) {
147            Ok(raw) => match serde_json::from_str::<serde_json::Value>(&raw) {
148                Ok(serde_json::Value::Object(map)) => Ok(map
149                    .into_iter()
150                    .filter_map(|(k, v)| v.as_str().map(|t| (k, t.to_string())))
151                    .collect()),
152                Ok(_) => Err(format!("{}: expected a JSON object", self.path.display())),
153                Err(e) => Err(format!("{}: {e}", self.path.display())),
154            },
155            Err(e) if e.kind() == std::io::ErrorKind::NotFound => Ok(BTreeMap::new()),
156            Err(e) => Err(format!("{}: {e}", self.path.display())),
157        }
158    }
159
160    fn store(&self, map: &BTreeMap<String, String>) -> Result<(), String> {
161        if let Some(dir) = self.path.parent() {
162            std::fs::create_dir_all(dir).map_err(|e| format!("create {}: {e}", dir.display()))?;
163        }
164        let tmp = self
165            .path
166            .with_extension(format!("tmp-{}", std::process::id()));
167        let body = serde_json::to_string(&serde_json::json!(map))
168            .map_err(|e| format!("encode credentials: {e}"))?;
169        std::fs::write(&tmp, body).map_err(|e| format!("write {}: {e}", tmp.display()))?;
170        #[cfg(unix)]
171        {
172            use std::os::unix::fs::PermissionsExt;
173            std::fs::set_permissions(&tmp, std::fs::Permissions::from_mode(0o600))
174                .map_err(|e| format!("chmod {}: {e}", tmp.display()))?;
175        }
176        std::fs::rename(&tmp, &self.path)
177            .map_err(|e| format!("rename {}: {e}", self.path.display()))
178    }
179}
180
181impl NodeCredentials for FileCredentials {
182    fn get(&self, node_id: &str) -> Result<Option<String>, String> {
183        Ok(self.load()?.get(node_id).cloned())
184    }
185
186    fn set(&self, node_id: &str, token: &str) -> Result<(), String> {
187        let _guard = self
188            .write
189            .lock()
190            .unwrap_or_else(std::sync::PoisonError::into_inner);
191        let mut map = self.load()?;
192        map.insert(node_id.to_string(), token.to_string());
193        self.store(&map)
194    }
195
196    fn delete(&self, node_id: &str) -> Result<(), String> {
197        let _guard = self
198            .write
199            .lock()
200            .unwrap_or_else(std::sync::PoisonError::into_inner);
201        let mut map = self.load()?;
202        map.remove(node_id);
203        self.store(&map)
204    }
205
206    fn backend(&self) -> &'static str {
207        "file"
208    }
209}

213fn file_store(hub_home: &Path) -> Arc<dyn NodeCredentials> {
214    Arc::new(FileCredentials::new(hub_home.join(FILE_NAME)))
215}
216
217/// Pick the backend from `std::env` — see [`from_lookup`].
218pub fn open(hub_home: &Path) -> Arc<dyn NodeCredentials> {
219    from_lookup(hub_home, |k| std::env::var(k).ok())
220}
221
222/// Selection logic with the env injected (test seam):
223/// - `file` — always the JSON file.
224/// - `keyring` — the OS keyring, even if the probe fails (explicit
225///   choice; per-call errors then surface honestly).
226/// - `auto`/unset/unknown — keyring when the probe passes, file else.
227pub fn from_lookup(
228    hub_home: &Path,
229    get: impl Fn(&str) -> Option<String>,
230) -> Arc<dyn NodeCredentials> {
231    pick(
232        hub_home,
233        get(STORE_ENV).as_deref().map(str::trim),
234        keyring_probe(),
235    )
236}
237
238/// The pick itself — env value and probe result injected so every arm
239/// is directly testable (a real probe can't be failed on demand).
240fn pick(
241    hub_home: &Path,
242    forced: Option<&str>,
243    probe: Result<(), String>,
244) -> Arc<dyn NodeCredentials> {
245    match forced {
246        Some("file") => file_store(hub_home),
247        Some("keyring") => {
248            if let Err(e) = &probe {
249                tracing::warn!("HYPHA_CREDENTIAL_STORE=keyring but the probe failed: {e}");
250            }
251            Arc::new(KeyringCredentials)
252        }
253        other => {
254            if !matches!(other, None | Some("" | "auto")) {
255                tracing::warn!(
256                    "unknown {STORE_ENV}={other:?} — expected auto|keyring|file; using auto"
257                );
258            }
259            match probe {
260                Ok(()) => Arc::new(KeyringCredentials),
261                Err(e) => {
262                    // Formatted eagerly — inside `info!`'s lazy arg
263                    // scope it only evaluates when a subscriber is live.
264                    let fallback = hub_home.join(FILE_NAME).display().to_string();
265                    tracing::info!(
266                        "OS keyring unavailable ({e}) — node credentials fall back to {fallback}"
267                    );
268                    file_store(hub_home)
269                }
270            }
271        }
272    }
273}

280    #[test]
281    fn pick_arms() {
282        let home = std::path::Path::new("/tmp/x");
283        // A thread-local subscriber enables the log macros' lazy arg
284        // evaluation (`fallback.display()` sits inside `info!`'s event
285        // scope) — a global `try_init` races subscriber registration
286        // in parallel tests, so scope it to the Err-probe calls.
287        let sub = tracing_subscriber::fmt()
288            .with_max_level(tracing::level_filters::LevelFilter::INFO)
289            .finish();
290        tracing::subscriber::with_default(sub, || {
291            // forced keyring with a failed probe → warn arm.
292            let _ = pick(home, Some("keyring"), Err("no dbus".into()));
293            // unknown value → warn + auto pick; probe-fail → file arm.
294            let _ = pick(home, Some("bogus"), Err("no dbus".into()));
295            let _ = pick(home, None, Err("no dbus".into()));
296        });
297        // A no-op dispatcher → the log macros' disabled paths.
298        tracing::dispatcher::with_default(&tracing::Dispatch::none(), || {
299            let _ = pick(home, Some("keyring"), Err("no dbus".into()));
300            let _ = pick(home, None, Err("no dbus".into()));
301        });
302        // forced keyring with a live probe → the `if let Err` skip arm.
303        let _ = pick(home, Some("keyring"), Ok(()));
304        let _ = pick(home, Some("auto"), Ok(()));
305        let _ = pick(home, Some("file"), Ok(()));
306        let _ = pick(home, None, Ok(()));
307    }
