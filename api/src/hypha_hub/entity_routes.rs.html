1//! The substrate's HTTP entity surface — `/api/entities*` on the hub.
2//! Reads hit the merged projection (every paired node), ops go through
3//! `SyncEntityApi::submit_op` so they queue in the durable outbox when
4//! the owning node is down, and snapshots route to the owning/target
5//! node. These handlers are registered ahead of the `/api/{*rest}`
6//! passthrough so the merged surface wins.
7

11use axum::response::{IntoResponse, Json, Response};
12use hypha_node_api::dto::EntityOp;
13use serde_json::{Value, json};

15use crate::HubState;

17fn json_response(body: Value, status: StatusCode) -> Response {
18    (status, Json(body)).into_response()
19}

21/// `EntityApi` errors → status codes: not-found 404, unregistered
22/// kinds/ops 501, lock contention 409, everything else 500.
23fn api_err(e: &str) -> Response {
24    let status = if e.contains("not found") || e.contains("Entity not found") {
25        StatusCode::NOT_FOUND
26    } else if e.contains("not registered") || e.contains("unknown op") {
27        StatusCode::NOT_IMPLEMENTED
28    } else if e.contains("held by") || e.contains("held on") {
29        StatusCode::CONFLICT
30    } else {
31        StatusCode::INTERNAL_SERVER_ERROR
32    };
33    json_response(json!({ "error": e }), status)
34}
35
36fn bad_request(msg: &str) -> Response {
37    json_response(json!({ "error": msg }), StatusCode::BAD_REQUEST)
38}

40fn query_param(req: &Request, key: &str) -> Option<String> {
41    req.uri().query().and_then(|q| {
42        q.split('&').find_map(|p| {
43            p.split_once('=')
44                .and_then(|(k, v)| if k == key { Some(v.to_string()) } else { None })
45        })
46    })
47}

49/// `GET /api/entities[?kind=][&node=]` — the merged entity index.
50/// `POST /api/entities` — create on the named (or primary) node.
51pub async fn collection(
52    State(state): State<HubState>,
53    method: Method,
54    req: Request<Body>,
55) -> Response {
56    match method {
57        Method::GET => {
58            let kind = query_param(&req, "kind");
59            match state.node.list_entities(kind.as_deref()).await {
60                Ok(entities) => json_response(json!({ "entities": entities }), StatusCode::OK),
61                Err(e) => api_err(&e),
62            }
63        }
64        Method::POST => {
65            let node_q = query_param(&req, "node");
66            let body = match read_body(req).await {
67                Ok(b) => b,
68                Err(res) => return *res,
69            };
70            // Explicit `kind` wins; else the api's registered default
71            // (first kind policy); a kind-less create on a backend
72            // with no default is a client error.
73            let kind = match body.get("kind").and_then(Value::as_str) {
74                Some(k) => k.to_string(),
75                None => match state.node.default_entity_kind() {
76                    Some(k) => k.to_string(),
77                    None => return bad_request("entity kind is required"),
78                },
79            };
80            let kind = kind.as_str();
81            let init = body
82                .get("indexDoc")
83                .cloned()
84                .unwrap_or_else(|| body.clone());
85            let node = body
86                .get("node")
87                .and_then(Value::as_str)
88                .map(str::to_string)
89                .or(node_q);
90            match state.node.create_entity(kind, &init, node.as_deref()).await {
91                Ok(res) => json_response(json!(res), StatusCode::CREATED),
92                Err(e) => api_err(&e),
93            }
94        }
95        _ => not_found(),
96    }
97}

99/// `GET /api/entities/{id}` — one entity doc. `DELETE` → `entity.delete`.
100pub async fn item(
101    State(state): State<HubState>,
102    method: Method,
103    Path(id): Path<String>,
104    _req: Request<Body>,
105) -> Response {
106    match method {
107        Method::GET => match state.node.entity(&id).await {
108            Ok(entity) => json_response(json!(entity), StatusCode::OK),
109            Err(e) => api_err(&e),
110        },
111        Method::DELETE => match state.node.delete_entity(&id).await {
112            Ok(()) => json_response(json!({ "ok": true }), StatusCode::OK),
113            Err(e) => api_err(&e),
114        },
115        _ => not_found(),
116    }
117}

119/// `POST /api/entities/{id}/ops` — `{kind, payload}` through the outbox.
120pub async fn ops(
121    State(state): State<HubState>,
122    method: Method,
123    Path(id): Path<String>,
124    req: Request<Body>,
125) -> Response {
126    if method != Method::POST {
127        return not_found();
128    }
129    let body = match read_body(req).await {
130        Ok(b) => b,
131        Err(res) => return *res,
132    };
133    let Some(kind) = body.get("kind").and_then(Value::as_str) else {
134        return bad_request("op kind is required");
135    };
136    let op = EntityOp {
137        kind: kind.to_string(),
138        payload: body.get("payload").cloned().unwrap_or_else(|| json!({})),
139    };
140    match state.node.submit_op(&id, &op).await {
141        Ok(()) => json_response(json!({ "ok": true }), StatusCode::ACCEPTED),
142        Err(e) => api_err(&e),
143    }
144}

146/// `GET /api/entities/{id}/snapshot` — export from the owning node.
147pub async fn snapshot_item(
148    State(state): State<HubState>,
149    method: Method,
150    Path(id): Path<String>,
151    _req: Request<Body>,
152) -> Response {
153    if method != Method::GET {
154        return not_found();
155    }
156    match state.node.snapshot_export(&id).await {
157        Ok(snapshot) => json_response(snapshot, StatusCode::OK),
158        Err(e) => api_err(&e),
159    }
160}

162/// `POST /api/entities/snapshot` — import on `?node=` (or the primary).
163pub async fn snapshot_collection(
164    State(state): State<HubState>,
165    method: Method,
166    req: Request<Body>,
167) -> Response {
168    if method != Method::POST {
169        return not_found();
170    }
171    let node = query_param(&req, "node");
172    let body = match read_body(req).await {
173        Ok(b) => b,
174        Err(res) => return *res,
175    };
176    match state.node.snapshot_import(&body, node.as_deref()).await {
177        Ok(entity) => json_response(json!(entity), StatusCode::CREATED),
178        Err(e) => api_err(&e),
179    }
180}

193/// `GET /api/pending-ops` — outbox depth: queued + dead-letter ops.
194pub async fn pending_ops(State(state): State<HubState>, method: Method) -> Response {
195    if method != Method::GET {
196        return not_found();
197    }
198    match state.node.pending_ops().await {

200        Err(e) => api_err(&e),
201    }

204/// Body → JSON object (`{}` when absent/empty); oversized/invalid →
205/// the boxed response the handler returns early.
206async fn read_body(req: Request<Body>) -> Result<Value, Box<Response>> {
207    const CAP: usize = 64 * 1024;
208    let bytes = axum::body::to_bytes(req.into_body(), CAP)
209        .await
210        .map_err(|_| Box::new((StatusCode::PAYLOAD_TOO_LARGE, "body too large").into_response()))?;
211    if bytes.is_empty() {
212        return Ok(json!({}));
213    }
214    serde_json::from_slice(&bytes).map_err(|_| Box::new(bad_request("Expected a JSON object body")))
215}
216
217fn not_found() -> Response {
218    json_response(json!({ "error": "not found" }), StatusCode::NOT_FOUND)
