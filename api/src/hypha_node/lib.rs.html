1//! hypha-node — the node host library: the thin composition layer
2//! between [`hypha_http::Env`] and the substrate axum app.
3//!
4//! A bare `hypha serve` mounts zero entity kinds — the substrate
5//! surface (`/api/health`, `/api/node`, `/api/events`, `/api/pair`,
6//! `/api/auth/*`, `/api/user`, `/api/fs`, `/api/push/*`,
7//! `/api/entity-index`, `/api/pending-ops`, and the `entity.*` ops
8//! namespace) still works end-to-end. Integrations
9//! (peregrin's session plane, campaign planes, …) compose
10//! [`state`] themselves: build a [`HyphaState`], register their
11//! [`EntityPlane`]s via [`HyphaState::with_kind`], and hand it to
12//! [`serve_state`].
13
14use std::net::SocketAddr;
15
16use hypha_http::{Env, HyphaState, app};
17
18/// The `HyphaState` a bare node serves — [`HyphaState::from_env`]
19/// with no kind planes registered. Integrations wanting planes call
20/// `HyphaState::from_env(env)` directly and `.with_kind(...)`.
21pub fn state(env: &Env) -> HyphaState {
22    HyphaState::from_env(env)
23}
24
25/// Build state from env and serve until shutdown — the whole
26/// `hypha serve` boot path. Call on a tokio runtime; returns when the
27/// listener fails or after graceful shutdown (SIGINT/SIGTERM).
28///
29/// # Errors
30/// `std::io::Error` on bind or serve failure.
31pub async fn serve(env: &Env) -> std::io::Result<()> {
32    serve_state(env, state(env)).await
33}
34
35/// Serve a pre-composed [`HyphaState`] — the integration entry point.
36/// Same bind + graceful-shutdown contract as [`serve`].
37///
38/// # Errors
39/// `std::io::Error` on bind or serve failure.
40pub async fn serve_state(env: &Env, state: HyphaState) -> std::io::Result<()> {
41    let listener = tokio::net::TcpListener::bind((env.host.as_str(), env.port)).await?;
42    tracing::info!("hypha-node listening on {}:{}", env.host, env.port);
43    // `ConnectInfo` feeds the /api/pair limiter its client-IP key.
44    let serving = axum::serve(
45        listener,
46        app(state).into_make_service_with_connect_info::<SocketAddr>(),
47    );
48    #[cfg(unix)]
49    {
50        use tokio::signal::unix::{SignalKind, signal};
51        let mut term = signal(SignalKind::terminate())?;
52        tokio::select! {
53            r = serving => r,
54            _ = tokio::signal::ctrl_c() => {
55                tracing::info!("hypha-node shutting down");
56                Ok(())
57            }
58            _ = term.recv() => {
59                tracing::info!("hypha-node shutting down");
60                Ok(())
61            }
62        }
63    }
64    #[cfg(not(unix))]
65    {
66        serving
67            .with_graceful_shutdown(async {
68                let _ = tokio::signal::ctrl_c().await;
69                tracing::info!("hypha-node shutting down");
70            })
71            .await
72    }
73}
