1//! Test fixtures — a deterministic in-memory [`EntityApi`]. `FakeEntityApi::seeded()`
2//! returns a fixed world (three entities across two kinds, a node
3//! descriptor, an empty outbox); `.with_*` knobs override individual
4//! fixtures and `.with_error(..)` makes every read endpoint fail.
5

17/// An `EntityApi` that answers every read from in-memory fixtures.
18/// Mutations are no-ops returning plausible successes.
19#[derive(Clone)]
20pub struct FakeEntityApi {
21    entities: Vec<EntityDto>,
22    node_info: NodeInfoDto,
23    nodes: Vec<NodeStatusDto>,
24    pending: Vec<PendingOpDto>,
25    error: Option<String>,
26}
27
28impl Default for FakeEntityApi {
29    fn default() -> Self {
30        Self::seeded()
31    }
32}
33

82    /// Replace the entity list.
83    #[must_use]
84    pub fn with_entities(mut self, entities: Vec<EntityDto>) -> Self {
85        self.entities = entities;
86        self
87    }
88
89    /// Replace the pending-ops fixture.
90    #[must_use]
91    pub fn with_pending(mut self, pending: Vec<PendingOpDto>) -> Self {
92        self.pending = pending;
93        self
94    }
95
96    /// Every read fails with `msg`.
97    #[must_use]
98    pub fn with_error(mut self, msg: impl Into<String>) -> Self {
99        self.error = Some(msg.into());
100        self
101    }
102
103    /// Wrap in the shared trait object shape callers expect.
104    pub fn shared(self) -> Arc<dyn EntityApi> {
105        Arc::new(self)
106    }
107
108    fn fail<T>(&self, v: T) -> Result<T, String> {
109        self.error.clone().map_or(Ok(v), Err)
110    }
111}

113#[async_trait::async_trait]
114impl EntityApi for FakeEntityApi {
115    async fn list_entities(&self, kind: Option<&str>) -> Result<Vec<EntityDto>, String> {
116        self.fail(
117            self.entities
118                .iter()
119                .filter(|e| kind.is_none_or(|k| e.kind == k))
120                .cloned()
121                .collect(),
122        )
123    }
124    async fn entity(&self, id: &str) -> Result<EntityDto, String> {
125        if let Some(e) = &self.error {
126            return Err(e.clone());
127        }
128        self.entities
129            .iter()
130            .find(|e| e.id == id)
131            .cloned()
132            .ok_or_else(|| format!("entity {id} not found"))
133    }
134    async fn create_entity(
135        &self,
136        kind: &str,

138        _node: Option<&str>,

143                id: "e-new".into(),
144                kind: kind.to_string(),

146                ..EntityDto::default()
147            }),
148        })
149    }
