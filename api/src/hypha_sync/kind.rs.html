1//! Per-kind policy registry — what the sync layer knows about each
2//! entity kind beyond the generic `/api/entities*` contract.
3//!
4//! The substrate moves entity docs, ops, and snapshots opaquely; a
5//! [`KindPolicy`] carries the kind-specific knowledge the hub needs
6//! to do its job: which SSE diff events project the kind, how
7//! tombstones resolve, how entity-scoped calls map onto node wire
8//! params/stream paths, which ops contend for the entity lock, how
9//! outbox replays classify, how a locked row names its external
10//! holder, and how wire docs columnize. Integrations register one
11//! policy per kind at composition; unregistered kinds get the generic
12//! behavior everywhere.
13//!
14//! Nothing here names a session, campaign, or orchestration type —
15//! policies are data + small functions over `serde_json::Value`.
16

23/// Column values a wire entity doc projects to. Every field is
24/// optional — `None` falls back to the generic extraction (the
25/// `{id,kind,indexDoc}` wrap unwrap plus the tolerant field reads).
26#[derive(Clone, Debug, Default)]
27pub struct IndexColumns {
28    /// Entity id — defaults to `id` on the doc (or inside `indexDoc`).
29    pub entity_id: Option<String>,
30    /// Entity kind tag — defaults to `kind` on the doc.
31    pub kind: Option<String>,
32    /// RFC 3339 `updatedAt`.
33    pub updated_at: Option<String>,
34    /// Lock-held flag.
35    pub locked: Option<bool>,
36    /// Live flag.
37    pub live: Option<bool>,
38}

40/// How a `{deleted: true}` index diff resolves for a kind's event.
41#[derive(Clone, Copy, Debug, PartialEq, Eq)]
42pub enum DiffTombstone {
43    /// Drop the projected entity row outright.
44    DeleteEntity,
45    /// Refetch the entity doc — the row survives (e.g. an overlay was
46    /// removed but the entity lives on).
47    Refetch,
48}

50/// Notification channel a [`Notice`] targets — the host maps it onto
51/// its notifier (the hub's Web Push has exactly these two).
52#[derive(Clone, Copy, Debug, PartialEq, Eq)]
53pub enum NoticeKind {
54    /// Terminal/completion notice.
55    Done,
56    /// Human input requested.
57    Permission,
58}

60/// A user-facing notification a kind wants delivered for a diff —
61/// produced by [`KindPolicy::notices`], rendered by the host.
62#[derive(Clone, Debug, PartialEq, Eq)]
63pub struct Notice {
64    /// Channel/category.
65    pub kind: NoticeKind,
66    /// Notification title.
67    pub title: String,
68    /// Notification body.
69    pub body: String,
70    /// Deep-link the notification opens.
71    pub url: String,
72}

74/// One entity kind's sync-side policy. All methods have inert
75/// defaults — a minimal policy is `entity_kinds` plus whatever the
76/// kind actually needs.
77pub trait KindPolicy: Send + Sync {
78    /// Entity kind tags this policy serves (`"sepia-session"`,
79    /// `"campaign"`, …). A kind is "registered" for create/gating
80    /// purposes iff some policy claims it.
81    fn entity_kinds(&self) -> &'static [&'static str];
82
83    /// SSE `event:` names whose `{id, patch}` frames are this kind's
84    /// index diffs. Empty = the kind only syncs via full listings.
85    fn diff_events(&self) -> &'static [&'static str] {
86        &[]
87    }
88
89    /// Tombstone resolution for one of [`Self::diff_events`].
90    fn tombstone(&self, _event: &str) -> DiffTombstone {
91        DiffTombstone::DeleteEntity
92    }
93
94    /// Does this policy arbitrate `op` (`<ns>.<verb>`)? First
95    /// registered claimant wins — keep namespaces disjoint.
96    fn owns_op(&self, _op: &str) -> bool {
97        false
98    }
99
100    /// Does `op` contend for the entity's live lock (gated writes
101    /// consult the hub's lock book before queueing/posting)? Only
102    /// consulted for ops [`Self::owns_op`] claims; `entity.delete`
103    /// is substrate-gated regardless.
104    fn op_gated(&self, _op: &str) -> bool {
105        false
106    }
107
108    /// Outbox replay classification for an owned op —
109    /// [`OpKind::Metadata`] retries, [`OpKind::Turn`] dead-letters.
110    /// Only consulted for ops [`Self::owns_op`] claims.
111    fn op_kind(&self, _op: &str) -> OpKind {
112        OpKind::Turn
113    }
114
115    /// Query params appended to entity-scoped node calls
116    /// (`GET /api/entities/{id}`, `POST …/ops`, stream opens).
117    /// `scope` is the caller's hint — today the entity kind value
118    /// itself, forwarded verbatim (the interim disambiguator).
119    fn call_params(&self, _scope: Option<&str>) -> Vec<(String, String)> {
120        Vec::new()
121    }
122
123    /// Live event-stream path for the entity — `id` is already
124    /// percent-encoded. `None` = the kind has no entity stream.
125    fn stream_path(&self, _id: &str) -> Option<String> {
126        None
127    }
128
129    /// The holder id recorded for a `locked` projection row the hub
130    /// never granted — `None` reads as `"external"`. Only consulted
131    /// for rows whose `kind` this policy serves.
132    fn external_actor_id(&self, _row: &IndexedEntity) -> Option<String> {
133        None
134    }
135
136    /// Column extraction override for this kind's wire docs — merged
137    /// over the generic rules field-by-field. `None` = generic.
138    fn columns(&self, _raw: &Value) -> Option<IndexColumns> {
139        None
140    }
141
142    /// The canonical entity kind a create request for `requested`
143    /// maps to — `None` = this policy doesn't serve the create.
144    /// Default: the kind itself when it's one of
145    /// [`Self::entity_kinds`]; integrations override to accept aliases
146    /// (`"session"` → `"sepia-session"`).
147    fn create_kind(&self, requested: &str) -> Option<&'static str> {
148        self.entity_kinds()
149            .iter()
150            .copied()
151            .find(|k| *k == requested)
152    }
153
154    /// Create-payload validation — runs after [`Self::create_kind`]
155    /// accepts the kind. `Err` rejects the create before it posts.
156    fn validate_create(&self, _init: &Value) -> Result<(), String> {
157        Ok(())
158    }
159
160    /// User-facing notices this kind wants delivered for a diff
161    /// `event` on `entity_id`. `row` is the projected entity when
162    /// indexed (`None` — e.g. a diff ahead of the first listing —
163    /// implementations fall back to the id). Default: none.
164    fn notices(
165        &self,
166        _event: &str,
167        _entity_id: &str,
168        _row: Option<&IndexedEntity>,
169        _patch: &Value,
170    ) -> Vec<Notice> {
171        Vec::new()
172    }
173}

175/// `kind`/`op`/`event` → [`KindPolicy`] registry. Registration order
176/// matters only for first-claim wins (`owns_op`, `diff_events`,
177/// `entity_kinds` lookups); keep namespaces disjoint.
178#[derive(Clone, Default)]
179pub struct KindPolicies {
180    inner: Arc<Vec<Arc<dyn KindPolicy>>>,
181}
182
183impl KindPolicies {
184    /// An empty registry — every lookup misses, every kind gets the
185    /// generic behavior.
186    pub fn new() -> Self {
187        Self::default()
188    }
189
190    /// Register a policy. Re-registering a policy whose
191    /// `entity_kinds` overlap replaces the earlier claimant in place
192    /// (recomposition is idempotent).
193    pub fn register(&mut self, policy: Arc<dyn KindPolicy>) {
194        let mut inner = Vec::clone(&self.inner);
195        let kinds = policy.entity_kinds();
196        inner.retain(|p| !p.entity_kinds().iter().any(|k| kinds.contains(k)));
197        inner.push(policy);
198        self.inner = Arc::new(inner);
199    }
200
201    /// Every registered policy, in registration order.
202    pub fn policies(&self) -> &[Arc<dyn KindPolicy>] {
203        &self.inner
204    }
205
206    pub fn is_empty(&self) -> bool {
207        self.inner.is_empty()
208    }
209
210    /// The policy serving entity `kind`, if registered.
211    pub fn for_entity_kind(&self, kind: &str) -> Option<Arc<dyn KindPolicy>> {
212        self.inner
213            .iter()
214            .find(|p| p.entity_kinds().contains(&kind))
215            .cloned()
216    }
217
218    /// The policy arbitrating `op`, if any claims it.
219    pub fn for_op(&self, op: &str) -> Option<Arc<dyn KindPolicy>> {
220        self.inner.iter().find(|p| p.owns_op(op)).cloned()
221    }
222
223    /// The diff-event resolution for a feed `event:` name — the
224    /// claiming policy plus its tombstone semantics. `None` = the
225    /// event isn't any registered kind's index diff (inert).
226    pub fn diff_policy(&self, event: &str) -> Option<(Arc<dyn KindPolicy>, DiffTombstone)> {
227        self.inner.iter().find_map(|p| {
228            p.diff_events()
229                .contains(&event)
230                .then(|| (Arc::clone(p), p.tombstone(event)))
231        })
232    }
233
234    /// The policy claiming diff `event` (one of its
235    /// [`KindPolicy::diff_events`]) — the feed-hook path. `None` = the
236    /// event isn't an index diff any kind projects.
237    pub fn for_diff_event(&self, event: &str) -> Option<Arc<dyn KindPolicy>> {
238        self.inner
239            .iter()
240            .find(|p| p.diff_events().contains(&event))
241            .cloned()
242    }
243
244    /// The canonical kind a create for `requested` maps to — first
245    /// policy whose [`KindPolicy::create_kind`] accepts wins.
246    pub fn create_kind(&self, requested: &str) -> Option<&'static str> {
247        self.inner.iter().find_map(|p| p.create_kind(requested))
248    }
249
250    /// The policy serving a create for `requested`, with the canonical
251    /// kind it maps to — `None` = no registered integration accepts
252    /// the create.
253    pub fn for_create(&self, requested: &str) -> Option<(Arc<dyn KindPolicy>, &'static str)> {
254        self.inner.iter().find_map(|p| {
255            p.create_kind(requested)
256                .map(|canonical| (Arc::clone(p), canonical))
257        })
258    }
259
260    /// The first registered kind — the default for kind-less creates
261    /// and imports (mirrors the node-side default plane).
262    pub fn default_entity_kind(&self) -> Option<&'static str> {
263        self.inner
264            .first()
265            .and_then(|p| p.entity_kinds().first().copied())
266    }
267
268    /// Column extraction for one wire entity doc: the kind's policy
269    /// override merged field-by-field over the generic rules. The
270    /// generic rules unwrap the `{indexDoc}` interim wrap and read
271    /// `updatedAt`/`locked`/`live` tolerantly (`live` also honors the
272    /// legacy `busy` alias) — see [`crate::columns_of`].
273    pub fn columns_of(&self, raw: &Value) -> IndexedEntity {
274        let mut row = crate::columns_of(raw);
275        if let Some(cols) = self.for_entity_kind(&row.kind).and_then(|p| p.columns(raw)) {
276            if let Some(v) = cols.entity_id {
277                row.entity_id = v;
278            }
279            if let Some(v) = cols.kind {
280                row.kind = v;
281            }
282            if let Some(v) = cols.updated_at {
283                row.updated_at = v;
284            }
285            if let Some(v) = cols.locked {
286                row.locked = v;
287            }
288            if let Some(v) = cols.live {
289                row.live = v;
290            }
291        }
292        row
293    }
