1//! hypha-sync — the hub-side sync layer (entity plane, ADR-0091).
2//!
3//! Keeps a local SQLite projection ([`ProjectionStore`]) of every paired
4//! node's entity index: `index_doc` rows only — opaque integration
5//! docs — never detail payloads (those stay node-fetched). The
6//! [`SyncEngine`] runs one supervised loop per node: connect →
7//! `list_entities` → drain the durable [`Outbox`] → tail `/api/events`
8//! for index diffs → reconnect with capped exponential backoff + jitter.
9//!
10//! Writes while a node is down go through [`SyncHandle::submit_op`],
11//! which posts directly when the node is up and enqueues into
12//! `hypha-outbox` otherwise.
13

30/// Failures of the sync layer — store, outbox, and node calls share one
31/// error type at the engine boundary.
32#[derive(Debug, thiserror::Error)]
33pub enum SyncError {
34    /// Projection-store failure.
35    #[error("sqlite: {0}")]
36    Sqlite(#[from] rusqlite::Error),
37    /// Outbox failure.
38    #[error("outbox: {0}")]
39    Outbox(#[from] hypha_outbox::OutboxError),
40    /// A node call failed (transport, HTTP status, malformed body).
41    #[error("node: {0}")]
42    Node(#[from] NodeError),
43    /// JSON (de)serialization failure.
44    #[error("serde: {0}")]
45    Serde(#[from] serde_json::Error),
46    /// The node id isn't registered with the engine.
47    #[error("unknown node {0}")]
48    UnknownNode(String),
49    /// The engine is already shut down (command channel closed).
50    #[error("sync engine is shut down")]
51    Shutdown,
52}

65/// Last known reachability of a node — written to `nodes.status`.
66#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
67#[serde(rename_all = "camelCase")]
68pub enum NodeStatus {
69    /// Never contacted since registration.
70    #[default]
71    Unknown,
72    /// A request (list/events) succeeded recently.
73    Up,
74    /// The last connection attempt failed or the SSE stream dropped.
75    Down,
76}
77
78impl NodeStatus {
79    /// The wire/storage spelling: `"unknown"`, `"up"`, `"down"`.
80    pub fn as_str(self) -> &'static str {
81        match self {
82            Self::Unknown => "unknown",
83            Self::Up => "up",
84            Self::Down => "down",
85        }
86    }
87
88    fn parse(s: &str) -> Self {
89        match s {
90            "up" => Self::Up,
91            "down" => Self::Down,
92            _ => Self::Unknown,
93        }
94    }
95}

108/// An `entities` row — one node's index doc for an entity. `raw` keeps
109/// the whole wire object; the projection columnizes only the fields it
110/// needs to merge/route on (`kind`, `locked`, `live`, `updatedAt`) —
111/// the doc's *shape* is the integration's, never the substrate's.
112#[derive(Clone, Debug, PartialEq)]
113pub struct IndexedEntity {
114    pub node_id: String,
115    pub entity_id: String,
116    /// Entity kind tag (`sepia-session`, `campaign`, …).
117    pub kind: String,
118    /// RFC 3339 `updatedAt` from the node.
119    pub updated_at: String,
120    pub locked: bool,
121    /// A live driver/agent is attached (`live` on the wire).
122    pub live: bool,
123    /// The full entity doc `Value` the node sent (the `indexDoc`).
124    pub raw: Value,
125}

169/// Pull the column values out of one wire entity doc — unknown/missing
170/// fields degrade to defaults rather than failing the row. Docs that
171/// carry `indexDoc` (the `{id,kind,indexDoc}` entity wrap) derive
172/// their fields from inside it; kind-specific aliases (`busy` →
173/// `live`) are the kind policy's [`KindPolicy::columns`] override.
174pub(crate) fn columns_of(raw: &Value) -> IndexedEntity {
175    let doc = match raw.get("indexDoc") {
176        Some(d) if d.is_object() => d,
177        _ => raw,
178    };
179    let get = |k: &str| doc.get(k).and_then(Value::as_str).unwrap_or_default();
180    let flag = |k: &str| doc.get(k).and_then(Value::as_bool).unwrap_or(false);
181    IndexedEntity {
182        node_id: String::new(),
183        entity_id: raw
184            .get("id")
185            .or_else(|| doc.get("id"))
186            .and_then(Value::as_str)
187            .unwrap_or_default()
188            .to_string(),
189        kind: raw
190            .get("kind")
191            .and_then(Value::as_str)
192            .unwrap_or_default()
193            .to_string(),
194        updated_at: get("updatedAt").to_string(),
195        locked: flag("locked"),
196        live: flag("live"),
197        raw: raw.clone(),
198    }
199}

201fn write_row(
202    conn: &Connection,
203    node_id: &str,
204    doc: &Value,
205    policies: &crate::kind::KindPolicies,
206) -> Result<(), SyncError> {
207    let mut row = policies.columns_of(doc);
208    row.node_id = node_id.to_string();
209    conn.execute(
210        "INSERT INTO entities
211         (node_id, entity_id, kind, updated_at, locked, live, raw_json)
212         VALUES (?,?,?,?,?,?,?)
213         ON CONFLICT (node_id, entity_id) DO UPDATE SET
214            kind=excluded.kind, updated_at=excluded.updated_at,
215            locked=excluded.locked, live=excluded.live,
216            raw_json=excluded.raw_json",
217        params![
218            row.node_id,
219            row.entity_id,
220            row.kind,
221            row.updated_at,
222            row.locked,
223            row.live,
224            // `Value`'s `Display` is infallible compact JSON — no `?`
225            // arm to leave uncovered.
226            row.raw.to_string(),
227        ],
228    )?;
229    Ok(())
230}

232fn row_to_indexed(row: &rusqlite::Row<'_>) -> rusqlite::Result<IndexedEntity> {
233    let raw_json: String = row.get("raw_json")?;
234    let raw = serde_json::from_str(&raw_json).map_err(|e| {
235        rusqlite::Error::FromSqlConversionFailure(0, rusqlite::types::Type::Text, Box::new(e))
236    })?;
237    Ok(IndexedEntity {
238        node_id: row.get("node_id")?,
239        entity_id: row.get("entity_id")?,
240        kind: row.get("kind")?,
241        updated_at: row.get("updated_at")?,
242        locked: row.get::<_, i64>("locked")? != 0,
243        live: row.get::<_, i64>("live")? != 0,
244        raw,
245    })
246}

261/// The local projection — a materialized view of every paired node's
262/// entity index plus per-node liveness and resync cursors. `Clone`
263/// shares one connection (same pattern as `hypha-outbox`).
264#[derive(Clone)]
265pub struct ProjectionStore {
266    conn: Arc<Mutex<Connection>>,
267    /// Kind-policy registry — column extraction consults it so a
268    /// kind's wire doc can override the generic column rules.
269    policies: crate::kind::KindPolicies,
270}
271
272impl ProjectionStore {
273    /// Open (and create) a projection at `path` — WAL + busy_timeout,
274    /// matching `hypha-outbox`'s durability posture.
275    ///
276    /// # Errors
277    /// On sqlite open/schema failure.
278    pub fn open(path: &Path) -> Result<Self, SyncError> {
279        if let Some(dir) = path.parent() {
280            std::fs::create_dir_all(dir).ok();
281        }
282        let conn = Connection::open(path)?;
283        restrict_file_perms(path);
284        conn.pragma_update(None, "journal_mode", "WAL")?;
285        conn.pragma_update(None, "busy_timeout", 5000)?;
286        conn.execute_batch(SCHEMA)?;
287        Ok(Self {
288            conn: Arc::new(Mutex::new(conn)),
289            policies: crate::kind::KindPolicies::new(),
290        })
291    }
292
293    /// Attach a kind-policy registry — column extraction consults it
294    /// on every write.
295    #[must_use]
296    pub fn with_kind_policies(mut self, policies: crate::kind::KindPolicies) -> Self {
297        self.policies = policies;
298        self
299    }

419    /// Replace a node's whole index: upsert every listed entity doc,
420    /// delete rows the list no longer reports (tombstones), and advance
421    /// the `sync_state` cursor to the max `updatedAt` seen.
422    ///
423    /// # Errors
424    /// On sqlite/serde failure.
425    pub fn apply_listing(&self, node_id: &str, docs: &[Value]) -> Result<(), SyncError> {
426        let conn = self
427            .conn
428            .lock()
429            .unwrap_or_else(std::sync::PoisonError::into_inner);
430        let tx = conn.unchecked_transaction()?;
431        let mut keep = Vec::with_capacity(docs.len());
432        let mut max_updated = String::new();
433        for doc in docs {
434            write_row(&tx, node_id, doc, &self.policies)?;
435            if let Some(id) = doc.get("id").and_then(Value::as_str) {
436                keep.push(id.to_string());
437            }
438            if let Some(u) = doc.get("updatedAt").and_then(Value::as_str) {
439                if u > max_updated.as_str() {
440                    max_updated = u.to_string();
441                }
442            }
443        }
444        if keep.is_empty() {
445            tx.execute("DELETE FROM entities WHERE node_id=?", params![node_id])?;
446        } else {
447            // `IN` list — ids come from the node, not the user, but
448            // still go through bind params.
449            let marks = std::iter::repeat_n("?", keep.len())
450                .collect::<Vec<_>>()
451                .join(",");
452            let sql =
453                format!("DELETE FROM entities WHERE node_id=? AND entity_id NOT IN ({marks})");
454            let mut bind: Vec<&dyn rusqlite::ToSql> = vec![&node_id];
455            bind.extend(keep.iter().map(|id| id as &dyn rusqlite::ToSql));
456            tx.execute(&sql, rusqlite::params_from_iter(bind.iter()))?;
457        }
458        if !max_updated.is_empty() {
459            tx.execute(
460                "INSERT INTO sync_state (node_id, cursor, updated_at) VALUES (?,?,?)
461                 ON CONFLICT (node_id) DO UPDATE SET cursor=excluded.cursor, updated_at=excluded.updated_at",
462                params![node_id, max_updated, now_rfc3339()],
463            )?;
464        }
465        tx.commit()?;
466        Ok(())
467    }

469    /// Merge a `{patch}` diff into an indexed row's `raw_json` and
470    /// re-derive its columns. [`PatchStatus::Missing`] when no row
471    /// exists — the caller should fetch the full entity doc.
472    ///
473    /// # Errors
474    /// On sqlite/serde failure.
475    pub fn apply_patch(
476        &self,
477        node_id: &str,
478        entity_id: &str,
479        patch: &Value,
480    ) -> Result<PatchStatus, SyncError> {
481        let conn = self
482            .conn
483            .lock()
484            .unwrap_or_else(std::sync::PoisonError::into_inner);
485        let raw_json: Option<String> = conn
486            .query_row(
487                "SELECT raw_json FROM entities WHERE node_id=? AND entity_id=?",
488                params![node_id, entity_id],
489                |r| r.get(0),
490            )
491            .ok();
492        let Some(raw_json) = raw_json else {
493            return Ok(PatchStatus::Missing);
494        };
495        let mut raw: Value = serde_json::from_str(&raw_json)?;
496        // Diffs patch the entity's fields — under the interim
497        // `indexDoc` wrap that's the inner doc, else the doc itself.
498        let target = match raw.get_mut("indexDoc") {
499            Some(d) if d.is_object() => d.as_object_mut(),
500            _ => raw.as_object_mut(),
501        };
502        if let (Some(obj), Some(patch_obj)) = (target, patch.as_object()) {
503            for (k, v) in patch_obj {
504                obj.insert(k.clone(), v.clone());
505            }
506        }
507        write_row(&conn, node_id, &raw, &self.policies)?;
508        Ok(PatchStatus::Applied)
509    }

511    /// Drop every row belonging to a node id that's no longer
512    /// registered — a renamed id (`node`→`main`), a node removed from
513    /// `PEREGRIN_NODES`, or the single-node shorthand id replaced by a
514    /// real one. Their entity rows have no sync loop anymore and
515    /// would serve dead entities forever (click → "node not found").
516    /// The store is a cache — a still-configured node resyncs itself.
517    ///
518    /// # Errors
519    /// On sqlite failure.
520    pub fn purge_unregistered(&self, keep: &[String]) -> Result<(), SyncError> {
521        let conn = self
522            .conn
523            .lock()
524            .unwrap_or_else(std::sync::PoisonError::into_inner);
525        if keep.is_empty() {
526            // No nodes configured — dropping everything would also
527            // wipe the visible list; only a bug could get here, keep
528            // the data and let the caller decide.
529            return Ok(());
530        }
531        let marks = std::iter::repeat_n("?", keep.len())
532            .collect::<Vec<_>>()
533            .join(",");
534        let bind: Vec<&dyn rusqlite::ToSql> =
535            keep.iter().map(|id| id as &dyn rusqlite::ToSql).collect();
536        let tx = conn.unchecked_transaction()?;
537        for table in ["entities", "sync_state"] {
538            tx.execute(
539                &format!("DELETE FROM {table} WHERE node_id NOT IN ({marks})"),
540                rusqlite::params_from_iter(bind.iter()),
541            )?;
542        }
543        tx.execute(
544            &format!("DELETE FROM nodes WHERE id NOT IN ({marks})"),
545            rusqlite::params_from_iter(bind.iter()),
546        )?;
547        tx.commit()?;
548        Ok(())
549    }

588    /// Every projected entity — `node_id` scopes to one node, `None`
589    /// lists all nodes (the hub's merged entity list).
590    ///
591    /// # Errors
592    /// On sqlite/serde failure.
593    pub fn entities(&self, node_id: Option<&str>) -> Result<Vec<IndexedEntity>, SyncError> {
594        let conn = self
595            .conn
596            .lock()
597            .unwrap_or_else(std::sync::PoisonError::into_inner);
598        if let Some(node_id) = node_id {
599            let mut stmt =
600                conn.prepare("SELECT * FROM entities WHERE node_id=? ORDER BY updated_at DESC")?;
601            let rows = stmt
602                .query_map(params![node_id], row_to_indexed)?
603                .collect::<Result<Vec<_>, _>>()?;
604            Ok(rows)
605        } else {
606            let mut stmt = conn.prepare("SELECT * FROM entities ORDER BY updated_at DESC")?;
607            let rows = stmt
608                .query_map([], row_to_indexed)?
609                .collect::<Result<Vec<_>, _>>()?;
610            Ok(rows)
611        }
612    }

615    /// listing — informational; today's resync is always a full list).
616    ///
617    /// # Errors
618    /// On sqlite failure.
619    pub fn cursor(&self, node_id: &str) -> Result<Option<String>, SyncError> {
620        let conn = self
621            .conn
622            .lock()
623            .unwrap_or_else(std::sync::PoisonError::into_inner);
624        let cursor = conn

626                "SELECT cursor FROM sync_state WHERE node_id=?",
627                params![node_id],

641            .conn
642            .lock()
643            .unwrap_or_else(std::sync::PoisonError::into_inner);
644        conn.execute(
645            "INSERT INTO sync_state (node_id, cursor, updated_at) VALUES (?,?,?)
646             ON CONFLICT (node_id) DO UPDATE SET cursor=excluded.cursor, updated_at=excluded.updated_at",
647            params![node_id, cursor, now_rfc3339()],
648        )?;
649        Ok(())
650    }
651}

653fn node_row(row: &rusqlite::Row<'_>) -> rusqlite::Result<NodeRow> {
654    Ok(NodeRow {
655        id: row.get("id")?,
656        url: row.get("url")?,
657        label: row.get("label")?,
658        last_seen_at: row.get("last_seen_at")?,
659        status: NodeStatus::parse(&row.get::<_, String>("status")?),
660    })
661}

663/// A write destined for a node — the input to [`SyncHandle::submit_op`].
664/// When the node is down it becomes an [`OutboxEntry`]; `policy` decides
665/// retry-vs-dead-letter on replay failure.
666#[derive(Clone, Debug)]
667pub struct NodeWrite {
668    /// Target node (must be registered).
669    pub node_id: String,
670    /// Entity the write applies to.
671    pub entity_id: String,
672    /// Entity kind for `?kind=` disambiguation — the projection's `kind`
673    /// column is a good source.
674    pub entity_kind: Option<String>,
675    /// Outbox op name: `<kind>.<verb>` namespaced (`sepia.prompt`,
676    /// `entity.delete`, …).
677    pub op: String,
678    /// [`OpKind::Metadata`] retries; [`OpKind::Turn`] dead-letters on a
679    /// failed replay.
680    pub policy: OpKind,
681    /// The JSON body (empty object for `entity.delete` etc.).
682    pub payload: Value,
683    /// Client-supplied dedupe key — re-submitting the same key returns
684    /// the same queued entry instead of a duplicate write.
685    pub idempotency_key: String,
686    /// Seconds until the queued write expires (dead-letters); `None`
687    /// never expires.
688    pub ttl_seconds: Option<i64>,
689}
