1//! hypha-xtask — workspace automation.
2//!
3//! ```text
4//! cargo xtask check      # fmt --check + clippy -D warnings
5//! cargo xtask test       # cargo nextest run (falls back to cargo test)
6//! cargo xtask coverage   # llvm-cov report (extra args pass through)
7//! cargo xtask fmt        # cargo fmt --all
8//! ```
9
10use std::process::{Command, ExitCode};
11
12/// Run `program args…` inheriting stdio; returns true on success.
13fn run(program: &str, args: &[&str]) -> bool {
14    match Command::new(program).args(args).status() {
15        Ok(status) => status.success(),
16        Err(e) => {
17            eprintln!("xtask: cannot run {program}: {e}");
18            false
19        }
20    }
21}
22
23/// True when the `cargo-nextest` subcommand is installed.
24fn nextest_available() -> bool {
25    Command::new("cargo")
26        .args(["nextest", "--version"])
27        .stdout(std::process::Stdio::null())
28        .stderr(std::process::Stdio::null())
29        .status()
30        .is_ok_and(|s| s.success())
31}
32
33/// True when the `cargo-llvm-cov` subcommand is installed.
34fn llvm_cov_available() -> bool {
35    Command::new("cargo")
36        .args(["llvm-cov", "--version"])
37        .stdout(std::process::Stdio::null())
38        .stderr(std::process::Stdio::null())
39        .status()
40        .is_ok_and(|s| s.success())
41}
42
43fn main() -> ExitCode {
44    let mut args = std::env::args().skip(1);
45    let ok = match args.next().as_deref() {
46        Some("check") => {
47            run("cargo", &["fmt", "--all", "--", "--check"])
48                && run(
49                    "cargo",
50                    &[
51                        "clippy",
52                        "--workspace",
53                        "--all-targets",
54                        "--",
55                        "-D",
56                        "warnings",
57                    ],
58                )
59        }
60        Some("test") => {
61            let cmd: &[&str] = if nextest_available() {
62                &["nextest", "run", "--workspace"]
63            } else {
64                &["test", "--workspace"]
65            };
66            run("cargo", cmd)
67        }
68        // `cargo xtask coverage [llvm-cov args…]` — extra args pass
69        // through to cargo-llvm-cov (e.g. `--lcov --output-path
70        // lcov.info`); passing any report flag replaces the default
71        // --html.
72        Some("coverage") => {
73            if !llvm_cov_available() {
74                eprintln!(
75                    "xtask: cargo-llvm-cov not installed — \
76                     `cargo install cargo-llvm-cov` (needs the llvm-tools component)"
77                );
78                return ExitCode::FAILURE;
79            }
80            let extra: Vec<String> = args.collect();
81            let wants_report = extra.iter().any(|a| {
82                matches!(
83                    a.as_str(),
84                    "--text"
85                        | "--summary-only"
86                        | "--json"
87                        | "--lcov"
88                        | "--cobertura"
89                        | "--codecov"
90                        | "--open"
91                )
92            });
93            let html = if wants_report { "" } else { " --html" };
94            // Isolate coverage artifacts in a repo-local target dir —
95            // a shared CARGO_TARGET_DIR lets foreign profraws/objects
96            // land in this report at 0% and corrupt TOTAL.
97            let cov_target = std::env::current_dir()
98                .map(|d| d.join("target/llvm-cov-target"))
99                .unwrap_or_else(|_| "target/llvm-cov-target".into());
100            let cov = format!(
101                "eval \"$(cargo llvm-cov show-env --export-prefix)\" \
102                 && cargo llvm-cov --workspace{html} \"$@\""
103            );
104            let status = Command::new("sh")
105                .args(["-c", &cov, "xtask-coverage"])
106                .args(&extra)
107                .env("CARGO_LLVM_COV_TARGET_DIR", &cov_target)
108                .status();
109            match status {
110                Ok(s) => s.success(),
111                Err(e) => {
112                    eprintln!("xtask: cannot run coverage: {e}");
113                    false
114                }
115            }
116        }
117        Some("fmt") => run("cargo", &["fmt", "--all"]),
118        _ => {
119            eprintln!("usage: cargo xtask <check|test|coverage|fmt>");
120            return ExitCode::from(2);
121        }
122    };
123    if ok {
124        ExitCode::SUCCESS
125    } else {
126        ExitCode::FAILURE
127    }
128}
